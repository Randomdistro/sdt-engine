cmake_minimum_required(VERSION 3.24)
project(sdt_ec_investigation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
enable_testing()
get_filename_component(SDT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_subdirectory("${SDT_ROOT}/Release/SDT_Celestial_Three_Body_Solver" solver)
option(SDT_CANONICAL "Test release solver against frozen canonical laws" OFF)
if(SDT_CANONICAL)
  configure_file("${SDT_ROOT}/Engine/include/sdt/laws.hpp" "${CMAKE_BINARY_DIR}/canonical/sdt/laws.hpp" COPYONLY)
  target_include_directories(sdt-celestial-core BEFORE PUBLIC "${CMAKE_BINARY_DIR}/canonical")
  file(SHA256 "${SDT_ROOT}/Engine/include/sdt/laws.hpp" SDT_CURRENT_SHA)
  set_property(TARGET sdt-celestial-core PROPERTY COMPILE_DEFINITIONS
    SDT_LAWS_SHA256="${SDT_CURRENT_SHA}"
    SDT_SOLVER_COMPILER="${CMAKE_CXX_COMPILER_ID}-${CMAKE_CXX_COMPILER_VERSION}")
endif()
add_executable(sdt_ec_coupling sdt_ec_coupling.cpp)
target_include_directories(sdt_ec_coupling PRIVATE "${SDT_ROOT}/Engine/include")
target_link_libraries(sdt_ec_coupling PRIVATE sdt-celestial-core)
if(MSVC)
  target_compile_options(sdt_ec_coupling PRIVATE /W4 /permissive- /fp:strict /utf-8)
endif()
add_test(NAME ec-native-controls COMMAND sdt_ec_coupling)
