cmake_minimum_required(VERSION 3.24)
project(sdt_alpha_controls LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
get_filename_component(SDT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
file(SHA256 "${SDT_ROOT}/Engine/include/sdt/laws.hpp" LAWS_HASH)
add_executable(sdt_alpha_controls sdt_alpha_controls.cpp)
target_include_directories(sdt_alpha_controls PRIVATE "${SDT_ROOT}/Engine/include")
target_compile_definitions(sdt_alpha_controls PRIVATE SDT_LAWS_HASH="${LAWS_HASH}")
if(MSVC)
  target_compile_options(sdt_alpha_controls PRIVATE /W4 /permissive- /fp:strict /utf-8)
endif()
enable_testing()
add_test(NAME native-alpha-analytic-controls COMMAND sdt_alpha_controls)
